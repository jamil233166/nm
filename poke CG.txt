#include<stdio.h>
#include<conio.h>
#include<graphics.h>
#include<stdlib.h>
float round(float a);
int main()
{
int gd=DETECT,gm;
// gd=graphics driver (detects best graphics driver and assigns it as default,
// gm=graphics mode
int x1,y1,x2,y2,steps,k;
float xincr,yincr,x,y,dx,dy;
printf("enter x1,y1");
scanf("%d%d",&x1,&y1);
printf("enter x2,y2");
scanf("%d%d",&x2,&y2);
initgraph(&gd,&gm,"");
// initializes the graph
dx=x2-x1;
dy=y2-y1;
if(abs(dx)>abs(dy))
steps=abs(dx);
else
steps=abs(dy);
xincr=dx/steps;
yincr=dy/steps;
x=x1;
y=y1;
for(k=1;k<=steps;k++)
{
delay(100);
// for seeing the line drawing process slowly.
x+=xincr;
y+=yincr;
putpixel(round(x),round(y),WHITE);
}
outtextxy(200,20,"DDA");
// for printing text at desired screen location.
outtextxy(x1+5,y1-5,"(x1,y1)");
outtextxy(x2+5,y2+5,"(x2,y2)");
getch();
closegraph();
// closes the graph and comes back to previous graphic mode.
}
float round(float a)
{
int b=a+0.5;
return b;
}

#include<iostream>
#include<conio.h>
#include<graphics.h>
using namespace std;

void draw_circle(int xc,int yc,int r);
void symmetry(int xc,int yc,int x,int y);

int main()
{
    int xc,yc,r;
    int gd=DETECT,gm;

    initgraph(&gd,&gm,"");

    cout<<"Enter the center of the circle:"<<endl;
    cout<<"xc = ";
    cin>>xc;
    cout<<"yc = ";
    cin>>yc;
    cout<<"Enter the radius of the circle: ";
    cin>>r;

    draw_circle(xc,yc,r);

    getch();
    closegraph();
    return 0;
}

void draw_circle(int xc,int yc,int r)
{
    int x=0;
    int y=r;
    int p=1-r;

    symmetry(xc,yc,x,y);

    while(x<y)
    {
        x++;

        if(p<0)
        {
            p=p+2*x+3;
        }
        else
        {
            y--;
            p=p+2*(x-y)+5;
        }

        symmetry(xc,yc,x,y);
    }
}

void symmetry(int xc,int yc,int x,int y)
{
    putpixel(xc+x,yc+y,GREEN);
    putpixel(xc-x,yc+y,GREEN);
    putpixel(xc+x,yc-y,GREEN);
    putpixel(xc-x,yc-y,GREEN);
    putpixel(xc+y,yc+x,GREEN);
    putpixel(xc-y,yc+x,GREEN);
    putpixel(xc+y,yc-x,GREEN);
    putpixel(xc-y,yc-x,GREEN);
}

///Ellips generation algorithm


#include <iostream>
#include <graphics.h>
#include <conio.h>
#include <math.h>
using namespace std;

void drawEllipse(int xc, int yc, int rx, int ry)
{
    float x = 0;
    float y = ry;
    float rx2 = rx * rx;
    float ry2 = ry * ry;
    float dx = 2 * ry2 * x;
    float dy = 2 * rx2 * y;

    float p1 = ry2 - (rx2 * ry) + (0.25 * rx2);

    while (dx < dy)
    {
        putpixel(xc + x, yc + y, WHITE);
        putpixel(xc - x, yc + y, WHITE);
        putpixel(xc + x, yc - y, WHITE);
        putpixel(xc - x, yc - y, WHITE);

        if (p1 < 0)
        {
            x++;
            dx = dx + (2 * ry2);
            p1 = p1 + dx + ry2;
        }
        else
        {
            x++;
            y--;
            dx = dx + (2 * ry2);
            dy = dy - (2 * rx2);
            p1 = p1 + dx - dy + ry2;
        }
    }

    float p2 = ry2 * (x + 0.5) * (x + 0.5)
             + rx2 * (y - 1) * (y - 1)
             - rx2 * ry2;

    while (y >= 0)
    {
        putpixel(xc + x, yc + y, WHITE);
        putpixel(xc - x, yc + y, WHITE);
        putpixel(xc + x, yc - y, WHITE);
        putpixel(xc - x, yc - y, WHITE);

        if (p2 > 0)
        {
            y--;
            dy = dy - (2 * rx2);
            p2 = p2 + rx2 - dy;
        }
        else
        {
            y--;
            x++;
            dx = dx + (2 * ry2);
            dy = dy - (2 * rx2);
            p2 = p2 + dx - dy + rx2;
        }
    }
}

int main()
{
    int gd = DETECT, gm;
    initgraph(&gd, &gm, "");

    int xc, yc, rx, ry;

    cout << "Enter the center point:" << endl;
    cout << "xc = ";
    cin >> xc;

    cout << "yc = ";
    cin >> yc;

    cout << "Enter Rx and Ry:" << endl;
    cin >> rx >> ry;

    drawEllipse(xc, yc, rx, ry);

    getch();
    closegraph();
    return 0;
}

#include<stdio.h>
#include<conio.h>
#include<graphics.h>

void symmetry(int xc,int yc,int x,int y)
{
    putpixel(xc+x,yc+y,GREEN);
    putpixel(xc-x,yc+y,GREEN);
    putpixel(xc+x,yc-y,GREEN);
    putpixel(xc-x,yc-y,GREEN);
    putpixel(xc+y,yc+x,GREEN);
    putpixel(xc-y,yc+x,GREEN);
    putpixel(xc+y,yc-x,GREEN);
    putpixel(xc-y,yc-x,GREEN);
}

void draw_circle(int xc,int yc,int r)
{
    int x=0,y=r;
    int p=1-r;

    symmetry(xc,yc,x,y);

    while(x<y)
    {
        x++;

        if(p<0)
            p=p+2*x+1;
        else
        {
            y--;
            p=p+2*(x-y)+1;
        }

        symmetry(xc,yc,x,y);
    }
}

int main()
{
    int gd=DETECT,gm;
    int xc,yc,r;

    initgraph(&gd,&gm,"");

    printf("Enter center coordinates (xc, yc): ");
    scanf("%d%d",&xc,&yc);

    printf("Enter radius: ");
    scanf("%d",&r);

    draw_circle(xc,yc,r);

    getch();
    closegraph();
    return 0;
}
